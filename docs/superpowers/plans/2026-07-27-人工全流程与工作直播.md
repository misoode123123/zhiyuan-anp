# 人工全流程与工作直播 实施计划（里程碑 1）

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 让真人开发者在平台端到端走完「认领→为需求开工作台编码→提变更→测试→审批→发布→delivered」，且全工作过程对三方实时直播可视。

**Architecture:** 复用 `codews_session + requirement_id` 作为「worker 会话」（不新建表）；codews 工作台启动时绑定 requirement_id；新 `internal/worker/` 包做观察层——按 requirement_id 查 codews_session 行，经 codews 的 opencode live HTTP / claude·codex transcript reader 读消息，SSE 近实时推给三方。审批补「禁自批」。前端加指派入口与「工作直播」面板。

**Tech Stack:** Go（Gin + sqlx + PG golang-migrate）、Next.js（TS, App Router）、SSE（`text/event-stream`）。

## Global Constraints

- **PG only，禁 SQLite**（开发/生产）。单测用 PG（`.28` `anp_dev` 或测试库），不得回退 sqlite。
- **本机 go 命令前缀 `GOPATH=C:/Users/yxt/go`**（GOPATH 被污染，见记忆）。
- **全量回归 `go test -p 1 ./...`**（并发污染 anp_test 库，见记忆）。
- **开发完直接 `.28` 部署测试**，本机不跑功能测试（`.28` 是测试库，见记忆）。
- **RBAC 是集中式**：路由在 `internal/auth/guard.go` 的 `routeOps` 登记 → op，`auth/permission.go` 的 `OpRoles` 定 op→角色，`AutoRequire` 中间件统一拦；**不要**在 handler 里手写角色判断（self/owner 这类非角色逻辑才在 handler）。
- **commit**：conventional commits 中文（`feat/fix/docs/...`），body 每行 ≤100 字符，末尾 `Co-Authored-By: Claude <noreply@anthropic.com>`。
- **`change.approve` 角色 RBAC 已生效**（routeOps:26-27 → `change.approve` → `{dev,gatekeeper,admin}`），本计划**不动角色集**，只加「禁自批」。

```go
package codews

import (
	"encoding/json"
	"fmt"
	"net/http"
	"strings"
	"time"
)

// liveClient 读 opencode 工作台内置 API（带超时防卡死）。
var liveClient = &http.Client{Timeout: 3 * time.Second}

// LiveTranscript 读 opencode 活跃会话的归一化消息（HTTP 127.0.0.1:port/api/session/<id>/message）。
// 仅 opencode（claude/codex 走 ReaderFor 读磁盘）。无会话/读失败返回空（非致命）。
func LiveTranscript(port int, sessionID string) ([]TranscriptMsg, error) {
	if sessionID == "" {
		return nil, nil
	}
	resp, err := liveClient.Get(fmt.Sprintf("http://127.0.0.1:%d/api/session/%s/message", port, sessionID))
	if err != nil {
		return nil, err
	}
	defer resp.Body.Close()
	var r struct {
		Data []struct {
			Type  string `json:"type"`
			Parts []struct {
				Type string `json:"type"`
				Text string `json:"text"`
			} `json:"parts"`
		} `json:"data"`
	}
	if err := json.NewDecoder(resp.Body).Decode(&r); err != nil {
		return nil, err
	}
	var out []TranscriptMsg
	for _, msg := range r.Data {
		var sb strings.Builder
		for _, p := range msg.Parts {
			if p.Type == "text" && strings.TrimSpace(p.Text) != "" {
				sb.WriteString(p.Text)
			}
		}
		if sb.Len() == 0 {
			continue
		}
		out = append(out, TranscriptMsg{Role: msg.Type, Content: sb.String()})
	}
	return out, nil
}
```

```go
func (m *Manager) Ensure(psID, appID, repoDir, userID, toolName, reqID string) (*Session, error) {
	if toolName == "" {
		toolName = defaultTool
	}
	if userID == "" {
		userID = "anonymous"
	}
	key := appID + ":" + userID
	m.mu.Lock()
	tool, ok := m.tools[toolName]
	if !ok {
		m.mu.Unlock()
		return nil, fmt.Errorf("未知编码工具: %s（已注册: %v）", toolName, m.Tools())
	}
	// 同开发者同工具同需求 活跃 → 复用
	if s, exists := m.sessions[key]; exists && s.alive() && s.Tool == toolName && s.RequirementID == reqID {
		m.mu.Unlock()
		return s, nil
	}
	// 同开发者 换工具 或 换需求 → 停旧起新（换需求=新 opencode 会话，杜绝多需求串台）
	if old, exists := m.sessions[key]; exists && old.cmd != nil && old.cmd.Process != nil {
		_ = old.cmd.Process.Kill()
		delete(m.sessions, key)
	}
	port := m.allocPortLocked()
	if port == 0 {
		m.mu.Unlock()
		return nil, fmt.Errorf("无可用工作台端口(%d-%d)", portMin, portMax)
	}
	m.mu.Unlock()

	workDir := ensureWorktree(repoDir, userID)
	cmd, err := tool.Start(workDir, port, m.toolEnv(toolName))
	if err != nil {
		return nil, err
	}
	s := &Session{
		AppID: appID, UserID: userID, Tool: toolName, Port: port, RepoDir: repoDir, cmd: cmd, started: time.Now(),
		RequirementID: reqID,
		URL:           fmt.Sprintf("http://%s:%d", m.host, port),
	}
	m.mu.Lock()
	m.sessions[key] = s
	m.mu.Unlock()

	go func() {
		_ = cmd.Wait()
		if m.sessionLog != nil && s.logID != "" {
			_ = m.sessionLog.FinishSession(context.Background(), s.logID, m.finishCounts(s))
		}
		m.mu.Lock()
		if cur, ok := m.sessions[key]; ok && cur == s {
			delete(m.sessions, key)
		}
		m.mu.Unlock()
	}()

	if !waitListen(port, 6*time.Second) {
		return nil, fmt.Errorf("%s 工作台启动后未监听 :%d", toolName, port)
	}
	if toolName == "opencode" {
		s.SessionID = ensureSession(port, repoDir)
		if s.SessionID != "" {
			s.DeepURL = sessionDeepURL(s.URL, repoDir, s.SessionID)
		}
	}
	if m.sessionLog != nil {
		rec := &SessionRecord{ProjectSpaceID: psID, AppID: appID, UserID: userID, Tool: toolName, RepoDir: workDir, Port: port, SessionID: s.SessionID, RequirementID: reqID}
		if err := m.sessionLog.StartSession(context.Background(), rec); err == nil {
			s.logID = rec.ID
		} else {
			log.Printf("[codews] 落库 codews_session 失败(非致命): %v", err)
		}
	}
	return s, nil
}
```

`stream.go`:

```go
package worker

import (
	"fmt"
	"io"
	"time"

	"github.com/gin-gonic/gin"
)

// streamPollInterval SSE 后端轮询消息间隔（生产 1.5s；测试可调小）。
var streamPollInterval = 1500 * time.Millisecond

// Stream SSE：三方实时看某需求 worker 的编码会话消息增量。
func (h *Handler) Stream(c *gin.Context) {
	w, err := h.store.ActiveByRequirement(c.Request.Context(), c.Param("id"), c.Param("rid"))
	if err != nil {
		c.String(500, err.Error())
		return
	}
	if !allowSelfOrAdmin(c, w) {
		c.String(403, "无权限")
		return
	}
	c.Writer.Header().Set("Content-Type", "text/event-stream")
	c.Writer.Header().Set("Cache-Control", "no-cache")
	c.Writer.Header().Set("Connection", "keep-alive")
	flusher, _ := c.Writer.(http.Flusher) // 需 import net/http

	sent := 0
	deadline := time.Now().Add(30 * time.Minute) // 单流最长 30min，防泄漏
	for time.Now().Before(deadline) {
		msgs, rerr := ReadMessages(w)
		if rerr == nil && len(msgs) > sent {
			for _, m := range msgs[sent:] {
				fmt.Fprintf(c.Writer, "event: message\ndata: {\"role\":%q,\"content\":%q}\n\n", m.Role, m.Content)
				sent++
			}
			if flusher != nil {
				flusher.Flush()
			}
		}
		if w != nil && w.EndedAt != nil && sent >= len(msgs) {
			io.WriteString(c.Writer, "event: done\ndata: {}\n\n")
			if flusher != nil {
				flusher.Flush()
			}
			return
		}
		select {
		case <-c.Request.Context().Done():
			return
		case <-time.After(streamPollInterval):
		}
	}
}
```

（`http.Flusher` 用 `net/http`；补 import。`w` 在循环中不更新——若需反映新会话，每轮重新 `ActiveByRequirement`；MVP 用快照即可，可后续优化。）

```tsx
import { useEffect, useState } from "react";

type Msg = { role: string; content: string };

export function WorkerLive({ psId, reqId, token }: { psId: string; reqId: string; token: string }) {
  const [msgs, setMsgs] = useState<Msg[]>([]);
  const [active, setActive] = useState(false);

  useEffect(() => {
    if (!reqId) return;
    setMsgs([]);
    const es = new EventSource(
      `${API}/project-spaces/${psId}/requirements/${reqId}/worker-session/stream`
    );
    es.addEventListener("message", (e) => {
      try {
        setMsgs((m) => [...m, JSON.parse(e.data)]);
      } catch {}
    });
    es.addEventListener("done", () => setActive(false));
    es.onerror = () => setActive(false);
    // 先拉一次历史 + meta
    fetch(`${API}/project-spaces/${psId}/requirements/${reqId}/worker-session`, {
      headers: { Authorization: `Bearer ${token}` },
    })
      .then((r) => r.json())
      .then((j) => setActive(j.active));
    return () => es.close();
  }, [psId, reqId]);

  return (
    <div className="worker-live">
      <h4>工作直播 {active ? "🟢 进行中" : "⚪ 空闲"}</h4>
      <div className="chat">
        {msgs.map((m, i) => (
          <div key={i} className={m.role}>
            <b>{m.role}</b>: {m.content}
          </div>
        ))}
      </div>
    </div>
  );
}
```
