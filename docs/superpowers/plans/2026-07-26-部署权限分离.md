# 部署权限分离（dev 可部署 test 不可 prod）实现计划

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** dev 角色可自主部署/管理 test 环境，prod 上线仅 gatekeeper/admin 且必须过变更闸门；堵住"任何登录用户能 `/deploy env=prod` 直上线"的安全漏洞。

**Architecture:** RBAC 操作按 env 拆粒度（`app.deploy.test/prod` 等），`AutoRequire` 中间件把查询到的 roles 注入 gin context 供 env 敏感的 handler 自行判定；`Deploy` handler 对 `env=prod` 复用 `Promote` 的变更闸门 `HasApproved`；前端上线按钮统一走 `/promote`。

**Tech Stack:** Go(gin) + Next.js + PostgreSQL（禁 SQLite）

**Spec:** `docs/PRD/2026-07-26-部署权限分离test-prod-PRD.md`

import "github.com/gin-gonic/gin"

// rolesFromCtx 从 gin context 取 AutoRequire 注入的角色列表（env 敏感部署操作按 env 自行鉴权用）。
func rolesFromCtx(c *gin.Context) []string {
	if v, ok := c.Get("roles"); ok {
		if r, ok := v.([]string); ok {
			return r
		}
	}
	return nil
}

```go
func (h *Handler) Stop(c *gin.Context) {
	var in struct {
		Env string `json:"env"`
	}
	_ = c.ShouldBindJSON(&in)
	env := in.Env
	if !IsValidEnv(env) {
		env = EnvProd // 默认 prod（向后兼容现有调用）
	}
	if !auth.Allowed("app.stop."+env, rolesFromCtx(c)) {
		httpx.Err(c, 403, 40301, "无权限停止 "+env+" 实例")
		return
	}
	a, _ := h.store.Get(c.Request.Context(), c.Param("id"), c.Param("aid"))
	ins, _ := h.store.GetInstance(c.Request.Context(), c.Param("aid"), env) // env 替代硬编码 EnvProd
	if a == nil || ins == nil || ins.ContainerName == "" {
		httpx.Err(c, 400, 50020, "应用未在 "+env+" 部署")
		return
	}
	if _, err := h.deployer.Stop(c.Request.Context(), ins.ContainerName); err != nil {
		httpx.Err(c, 500, 50020, err.Error())
		return
	}
	_ = h.store.SetInstanceStatus(c.Request.Context(), a.ID, env, "stopped", "", "")
	if env == EnvProd {
		_ = h.store.SetStatus(c.Request.Context(), a.ProjectSpaceID, a.ID, "stopped", "", "")
	}
	httpx.OK(c, gin.H{"id": a.ID, "env": env, "status": "stopped"})
}
```
