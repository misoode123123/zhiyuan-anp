# 绩效记录 Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 按人自动统计开发产出 + 完整呈现开发者↔编码工具（opencode/claude/codex）的互动聊天记录，admin 看全员/本人看自己，顺手删考勤。

**Architecture:** 给 `code_task`/`change_request`/`conversation` 补可空 `user_id`（中间件多设 `CtxUserDBID`，零回归）；新增 `codews_session` 表存会话生命周期+计数；新增 `TranscriptReader` 接口读三工具原生磁盘 transcript（claude `~/.claude/projects`、codex `~/.codex/sessions`、opencode）归一化为统一消息；新 `internal/performance` 包跨表 `GROUP BY user_id` 按需聚合；API 走现有 `AutoRequire` RBAC（admin 端点登记 op，me 放行自校验）。

**Tech Stack:** Go + gin + sqlx + PostgreSQL（pgx）；前端 Next.js（app router）+ TypeScript。

**Spec:** `docs/PRD/2026-07-26-绩效记录-PRD.md`

## Global Constraints

- **只用 PG，禁 SQLite**（memory `no-sqlite-pg-only`）。所有 store 测试走 `testutil.TestDB(t)`（连 .28 `anp_test` 库，跑迁移建全表），`testutil.Truncate` 隔离；不写 sqlite `:memory:` 单测（memory `sqlite-test-pg-type-trap`）。
- **PG 占位符用 `$N`**；动态 WHERE 用 `sqlx.Rebind(sqlx.DOLLAR, q)`（`?`→`$N`）。
- **可空列 scan 必须容忍 NULL**（用 `COALESCE` 或指针字段；memory `deploy-prod-10.10.0.28` 记录的 system_config NULL 崩溃教训）。
- **多租户**：所有数据带 `project_space_id`；handler 从 `c.Param("id")` 或 `c.GetString("project_space_id")` 取。
- **RBAC**：写/危险操作登记到 `auth/guard.go:routeOps` + `auth/permission.go:permissions`，由 `AutoRequire` 统一校验；未登记路由默认放行。
- **开发完 scp 部署 .28 真打接口验证**（memory `deploy-28-no-local-test`、`verify-cross-frontend-backend`）；本机不跑功能测试。
- **方案先成文审核**已done（PRD `10c23df`）；本计划为实施依据。

```go
func (s *Store) ValidToken(ctx context.Context, token string) (string, string, bool) {
	if token == "" {
		return "", "", false
	}
	var name, uid string
	if err := s.db.GetContext(ctx, &name, uid,
		`SELECT user_name, user_id FROM auth_session WHERE token=$1 AND expires_at > CURRENT_TIMESTAMP`, token,

	}
	if err := s.db.GetContext(ctx, &row,
		`SELECT user_name, user_id FROM auth_session WHERE token=$1 AND expires_at > CURRENT_TIMESTAMP`, token,
```

```go
package codews

import (
	"context"
	"time"

	"github.com/google/uuid"
	"github.com/jmoiron/sqlx"
)

type SessionRecord struct {
	ID             string    `json:"id" db:"id"`
	ProjectSpaceID string    `json:"project_space_id" db:"project_space_id"`
	AppID          string    `json:"app_id" db:"app_id"`
	UserID         string    `json:"user_id,omitempty" db:"user_id"`
	Tool           string    `json:"tool" db:"tool"`
	RepoDir        string    `json:"repo_dir" db:"repo_dir"`
	Port           int       `json:"port" db:"port"`
	SessionID      string    `json:"session_id,omitempty" db:"session_id"`
	StartedAt      time.Time `json:"started_at" db:"started_at"`
	EndedAt        *time.Time `json:"ended_at,omitempty" db:"ended_at"`
	PromptCount    int       `json:"prompt_count" db:"prompt_count"`
	MessageCount   int       `json:"message_count" db:"message_count"`
}

type SessionCounts struct{ PromptCount, MessageCount int }

// SessionStore codews 会话持久化（落库供绩效/互动统计；nil=纯内存兼容）。
type SessionStore interface {
	StartSession(ctx context.Context, s *SessionRecord) error
	FinishSession(ctx context.Context, id string, counts SessionCounts) error
}

type pgSessionStore struct{ db *sqlx.DB }

func NewPGSessionStore(db *sqlx.DB) SessionStore { return &pgSessionStore{db: db} }

func (p *pgSessionStore) StartSession(ctx context.Context, s *SessionRecord) error {
	s.ID = "cws_" + uuid.NewString()[:20]
	_, err := p.db.ExecContext(ctx,
		`INSERT INTO codews_session (id, project_space_id, app_id, user_id, tool, repo_dir, port, session_id, started_at)
		 VALUES ($1,$2,$3,$4,$5,$6,$7,$8,CURRENT_TIMESTAMP)`,
		s.ID, s.ProjectSpaceID, s.AppID, nullable(s.UserID), s.Tool, s.RepoDir, s.Port, nullable(s.SessionID))
	return err
}

func (p *pgSessionStore) FinishSession(ctx context.Context, id string, c SessionCounts) error {
	_, err := p.db.ExecContext(ctx,
		`UPDATE codews_session SET ended_at=CURRENT_TIMESTAMP, prompt_count=$2, message_count=$3 WHERE id=$1`,
		id, c.PromptCount, c.MessageCount)
	return err
}

func nullable(s string) interface{} {
	if s == "" {
		return nil
	}
	return s
}
```

```go
// readerFor 按工具名返回 transcript reader（claude/codex 读磁盘；opencode Task 7 接入）。
func readerFor(tool string) TranscriptReader {
	switch tool {
	case "claude":
		return claudeReader{root: claudeHome()}
	case "codex":
		return codexReader{root: codexHome()}
	}
	return nil

func claudeHome() string {

		return filepath.Join(h, "projects")
	}
	if h, err := os.UserHomeDir(); err == nil {

	}
	return ".claude/projects"
}

func codexHome() string {

		return filepath.Join(h, "sessions")

	if h, err := os.UserHomeDir(); err == nil {

	}

// ---- 通用：扫 root 下所有 .jsonl，按 cwd 过滤行 ----

type rawLine struct {
	Type      string          `json:"type"`
	Role      string          `json:"role"`        // codex 用
	Message   json.RawMessage `json:"message"`     // claude 用
	Content   json.RawMessage `json:"content"`     // codex 用
	Cwd       string          `json:"cwd"`
	SessionID string          `json:"sessionId"`   // claude
	SessID2   string          `json:"session_id"`  // codex
	Timestamp string          `json:"timestamp"`
}

func scanJSONL(path string) []rawLine {
	f, err := os.Open(path)
	if err != nil {
		return nil
	}
	defer f.Close()
	var out []rawLine
	sc := bufio.NewScanner(f)
	sc.Buffer(make([]byte, 0, 64*1024), 4*1024*1024) // 单行最大 4MB
	for sc.Scan() {
		var l rawLine
		if json.Unmarshal(sc.Bytes(), &l) == nil {
			out = append(out, l)
		}
	}
	return out
}

// flattenContent 把 claude/codex 的 content（string 或 [{type,text}/{type,output_text}]）归一化为纯文本。
func flattenContent(raw json.RawMessage) string {
	if len(raw) == 0 {
		return ""
	}
	var s string
	if json.Unmarshal(raw, &s) == nil {
		return s
	}
	var arr []map[string]json.RawMessage
	if json.Unmarshal(raw, &arr) == nil {
		var sb strings.Builder
		for _, p := range arr {
			var txt string
			if v, ok := p["text"]; ok {
				_ = json.Unmarshal(v, &txt)
			} else if v, ok := p["output_text"]; ok {
				_ = json.Unmarshal(v, &txt)
			}
			if txt != "" {
				sb.WriteString(txt)
			}
		}
		return sb.String()
	}
	return ""
}

// ---- claude ----

type claudeReader struct{ root string }

func (r claudeReader) files() []string { return globJSONL(r.root) }

func (r claudeReader) Messages(repoDir, sessionID string) ([]TranscriptMsg, error) {
	var out []TranscriptMsg
	for _, p := range r.files() {
		for _, l := range scanJSONL(p) {
			if l.Type != "user" && l.Type != "assistant" {
				continue
			}
			if l.Cwd != repoDir || (sessionID != "" && l.SessionID != sessionID) {
				continue
			}
			var msg struct {
				Role    string          `json:"role"`
				Content json.RawMessage `json:"content"`
			}
			if json.Unmarshal(l.Message, &msg) != nil {
				continue
			}
			out = append(out, TranscriptMsg{Role: msg.Role, Content: flattenContent(msg.Content), CreatedAt: parseTS(l.Timestamp)})
		}
	}
	return out, nil
}

func (r claudeReader) Sessions(repoDir string) ([]TranscriptMeta, error) {
	byID := map[string]*TranscriptMeta{}
	for _, p := range r.files() {
		for _, l := range scanJSONL(p) {
			if l.Cwd != repoDir {
				continue
			}
			sid := l.SessionID
			if sid == "" {
				continue
			}
			m := byID[sid]
			if m == nil {
				m = &TranscriptMeta{SessionID: sid, Tool: "claude", Cwd: repoDir}
				byID[sid] = m
			}
			if t := parseTS(l.Timestamp); t.After(m.UpdatedAt) {
				m.UpdatedAt = t
			}
			if l.Type == "user" {
				m.PromptCount++
			}
		}
	}
	return metaList(byID), nil
}

func globJSONL(root string) []string {
	var files []string
	_ = filepath.Walk(root, func(p string, info os.FileInfo, err error) error {
		if err != nil || info.IsDir() {
			return nil
		}
		if strings.HasSuffix(p, ".jsonl") {
			files = append(files, p)
		}
		return nil
	})
	return files
}

func parseTS(s string) time.Time {
	if s == "" {
		return time.Time{}
	}
	for _, layout := range []string{time.RFC3339Nano, time.RFC3339, "2006-01-02T15:04:05Z"} {
		if t, err := time.Parse(layout, s); err == nil {
			return t
		}
	}
	return time.Time{}
}

func metaList(m map[string]*TranscriptMeta) []TranscriptMeta {
	out := make([]TranscriptMeta, 0, len(m))
	for _, v := range m {
		out = append(out, *v)
	}
	return out
```

```go
package performance

import (
	"context"
	"time"

	"github.com/jmoiron/sqlx"
)

type Store struct{ db *sqlx.DB }

func NewStore(db *sqlx.DB) *Store { return &Store{db: db} }

func (s *Store) Summary(ctx context.Context, psID, userID string, from, to time.Time) (Profile, error) {
	var p Profile
	p.UserID = userID
	// 名字
	_ = s.db.GetContext(ctx, &p.UserName, `SELECT name FROM "user" WHERE id=$1`, userID)

	w := window(from, to)

	// code_task
	if err := s.db.GetContext(ctx, &[]int{}, ``); err != nil {} // placeholder 删除
	var ct struct{ Done, Failed int }
	_ = s.db.GetContext(ctx, &ct,
		`SELECT COUNT(*) FILTER (WHERE status='completed') done,
		        COUNT(*) FILTER (WHERE status='failed') failed
		 FROM code_task WHERE project_space_id=$1 AND user_id=$2`+w.sql("created_at"), append([]interface{}{psID, userID}, w.args...)...)
	p.Metrics.CodeTaskDone, p.Metrics.CodeTaskFailed = ct.Done, ct.Failed

	// change_request（作者 = user_id；审批状态）
	var ch struct{ Sub, Appr, Rej int }
	_ = s.db.GetContext(ctx, &ch,
		`SELECT COUNT(*) sub,
		        COUNT(*) FILTER (WHERE status='approved') appr,
		        COUNT(*) FILTER (WHERE status='rejected') rej
		 FROM change_request WHERE project_space_id=$1 AND user_id=$2`+w.sql("created_at"), append([]interface{}{psID, userID}, w.args...)...)
	p.Metrics.ChangeSubmitted, p.Metrics.ChangeApproved, p.Metrics.ChangeRejected = ch.Sub, ch.Appr, ch.Rej

	// releases（join change 拿作者）
	_ = s.db.GetContext(ctx, &p.Metrics.Releases,
		`SELECT COUNT(*) FROM release_record r JOIN change_request c ON c.id=r.change_id
		 WHERE r.project_space_id=$1 AND c.user_id=$2`+w.sql("r.created_at"), append([]interface{}{psID, userID}, w.args...)...)

	// conversations
	_ = s.db.GetContext(ctx, &p.Metrics.Conversations,
		`SELECT COUNT(*) FROM conversation WHERE project_space_id=$1 AND user_id=$2`+w.sql("created_at"), append([]interface{}{psID, userID}, w.args...)...)

	// requirement 认领/完成
	var rq struct{ Claimed, Done int }
	_ = s.db.GetContext(ctx, &rq,
		`SELECT COUNT(*) claimed, COUNT(*) FILTER (WHERE status='done') done
		 FROM requirement WHERE project_space_id=$1 AND assignee=(SELECT name FROM "user" WHERE id=$2)`+w.sql("assigned_at"),
		append([]interface{}{psID, userID}, w.args...)...)
	p.Metrics.ReqClaimed, p.Metrics.ReqCompleted = rq.Claimed, rq.Done

	// 互动 codews_session
	var ws struct{ Sess, Prompts, Secs int }
	_ = s.db.GetContext(ctx, &ws,
		`SELECT COUNT(*) sess, COALESCE(SUM(prompt_count),0) prompts,
		        COALESCE(EXTRACT(EPOCH FROM (COALESCE(ended_at,NOW())-started_at))::int,0) secs
		 FROM codews_session WHERE project_space_id=$1 AND user_id=$2`+w.sql("started_at"),
		append([]interface{}{psID, userID}, w.args...)...)
	p.Metrics.WsSessions, p.Metrics.WsPrompts, p.Metrics.WsSeconds = ws.Sess, ws.Prompts, ws.Secs

	// 最近会话列表
	_ = s.db.SelectContext(ctx, &p.Sessions,
		`SELECT id, tool, repo_dir, started_at, ended_at, prompt_count
		 FROM codews_session WHERE project_space_id=$1 AND user_id=$2 ORDER BY started_at DESC LIMIT 20`, psID, userID)
	return p, nil
}

// Members 全员 + 未归属桶。
func (s *Store) Members(ctx context.Context, psID string, from, to time.Time) ([]Profile, error) {
	// 取空间内成员 user_id 并集（membership）∪ 各表出现的 user_id（含 NULL）
	var ids []struct {
		UID string `db:"uid"`
	}
	q := `SELECT user_id uid FROM membership WHERE project_space_id=$1 AND user_id IS NOT NULL
	      UNION SELECT user_id FROM code_task WHERE project_space_id=$2 AND user_id IS NOT NULL
	      UNION SELECT user_id FROM codews_session WHERE project_space_id=$3 AND user_id IS NOT NULL`
	if err := s.db.SelectContext(ctx, &ids, q, psID, psID, psID); err != nil {
		return nil, err
	}
	out := []Profile{}
	for _, id := range ids {
		p, _ := s.Summary(ctx, psID, id.UID, from, to)
		out = append(out, p)
	}
	// 未归属桶
	un, _ := s.unassignedSummary(ctx, psID, from, to)
	if un.Metrics.CodeTaskDone+un.Metrics.WsSessions+un.Metrics.ChangeSubmitted > 0 {
		un.IsUnassigned = true
		out = append(out, un)
	}
	return out, nil
}

func (s *Store) unassignedSummary(ctx context.Context, psID string, from, to time.Time) (Profile, error) {
	p := Profile{UserName: "未归属（历史无 user_id）"}
	w := window(from, to)
	var ct struct{ Done, Failed int }
	_ = s.db.GetContext(ctx, &ct,
		`SELECT COUNT(*) FILTER (WHERE status='completed') done, COUNT(*) FILTER (WHERE status='failed') failed
		 FROM code_task WHERE project_space_id=$1 AND user_id IS NULL`+w.sql("created_at"), append([]interface{}{psID}, w.args...)...)
	p.Metrics.CodeTaskDone, p.Metrics.CodeTaskFailed = ct.Done, ct.Failed
	var ws struct{ Sess, Prompts int }
	_ = s.db.GetContext(ctx, &ws,
		`SELECT COUNT(*), COALESCE(SUM(prompt_count),0) FROM codews_session WHERE project_space_id=$1 AND user_id IS NULL`+w.sql("started_at"),
		append([]interface{}{psID}, w.args...)...)
	p.Metrics.WsSessions, p.Metrics.WsPrompts = ws.Sess, ws.Prompts
	return p, nil
}

// window 时间窗辅助：from/to 零值=不加条件。
type windowArgs struct{ sql_ string; args []interface{} }
func (w windowArgs) sql(col string) string  { return w.sql_ }
func window(from, to time.Time) windowArgs {
	w := windowArgs{}
	if !from.IsZero() {
		w.sql_ += " AND " + "created_at_col >= ?"
	}
	return w
}
```

> ⚠️ 上面 `window` 辅助是**示意骨架有 bug**（`created_at_col` 不是绑定列名，且每查询列名不同）。**实现时改为**：定义 `func bound(col string, from, to time.Time) (clause string, args []interface{})`，返回 `" AND <col> >= $N AND <col> <= $N"` + 对应 args，在每个查询处单独调用并按已有占位计数续编号（因各 SQL 的 `$N` 基数不同，需就地拼）。或更简：**用 sqlx.Rebind + `?`**：每个查询写 `... AND created_at >= ? AND created_at <= ?`，args 追加 from/to（零值则不加该段），最后 `q = sqlx.Rebind(sqlx.DOLLAR, q)`。**采用 Rebind 方案**，重写上述各查询为 `?` 占位 + Rebind。实现时统一此法，删除 windowArgs 骨架。

```go
package performance

import (

	"zhiyuan-anp/platform/backend/internal/auth"
	"zhiyuan-anp/platform/backend/internal/codews"

)

type Handler struct {
	store *Store
	ws    *codews.Manager // opencode live messages（claude/codex 走 readerFor）
}

func NewHandler(store *Store, ws *codews.Manager) *Handler { return &Handler{store: store, ws: ws} }

func Register(r gin.IRouter, store *Store, ws *codews.Manager) {
	NewHandler(store, ws).Register(r)
}

func (h *Handler) Register(r gin.IRouter) {
	r.GET("/project-spaces/:id/performance/me", h.Me)
	r.GET("/project-spaces/:id/performance/members", h.Members)
	r.GET("/project-spaces/:id/performance/members/:userID", h.MemberDetail)
	r.GET("/project-spaces/:id/performance/sessions/:sessionID/messages", h.SessionMessages)
}

func parseWindow(c *gin.Context) (time.Time, time.Time) {
	from, _ := time.Parse("2006-01-02", c.Query("from"))
	to, _ := time.Parse("2006-01-02", c.Query("to"))
	if !to.IsZero() {
		to = to.Add(24 * time.Hour)
	}
	return from, to
}

// Me 本人绩效（自校验：取登录者 id）。
func (h *Handler) Me(c *gin.Context) {
	psID := c.Param("id")
	uid := c.GetString(auth.CtxUserDBID)
	if uid == "" {
		httpx.Err(c, 401, 40101, "未识别用户")
		return
	}
	from, to := parseWindow(c)
	p, err := h.store.Summary(c.Request.Context(), psID, uid, from, to)
	if err != nil {
		httpx.Err(c, 500, 50080, err.Error())
		return
	}
	c.JSON(200, p)
}

func (h *Handler) Members(c *gin.Context) {
	psID := c.Param("id")
	from, to := parseWindow(c)
	list, err := h.store.Members(c.Request.Context(), psID, from, to)
	if err != nil {
		httpx.Err(c, 500, 50080, err.Error())
		return
	}
	c.JSON(200, list)
}

func (h *Handler) MemberDetail(c *gin.Context) {
	psID := c.Param("id")
	uid := c.Param("userID")
	from, to := parseWindow(c)
	p, err := h.store.Summary(c.Request.Context(), psID, uid, from, to)
	if err != nil {
		httpx.Err(c, 500, 50080, err.Error())
		return
	}
	c.JSON(200, p)
}

// SessionMessages 某次互动的完整聊天记录（claude/codex 走磁盘 reader，opencode 走 live HTTP）。
func (h *Handler) SessionMessages(c *gin.Context) {
	sessionID := c.Param("sessionID")
	// 查 codews_session 取 tool/repo_dir/app_id/user_id
	rec, err := h.store.SessionByID(c.Request.Context(), sessionID)
	if err != nil {
		httpx.Err(c, 404, 40401, "会话不存在")
		return
	}
	// 分流：opencode live；claude/codex 磁盘 reader
	if rec.Tool == "opencode" && h.ws != nil {
		txt, err := h.ws.SessionMessages(rec.AppID, rec.UserID)
		if err != nil {
			httpx.Err(c, 500, 50081, err.Error())
			return
		}
		c.JSON(200, gin.H{"tool": "opencode", "transcript": txt})
		return
	}
	if r := codews.ReaderFor(rec.Tool); r != nil {
		msgs, err := r.Messages(rec.RepoDir, sessionID)
		if err != nil {
			httpx.Err(c, 500, 50081, err.Error())
			return
		}
		c.JSON(200, gin.H{"tool": rec.Tool, "messages": msgs})
		return
	}
	httpx.Err(c, 400, 40001, "该工具不支持查看聊天记录")
}
```
